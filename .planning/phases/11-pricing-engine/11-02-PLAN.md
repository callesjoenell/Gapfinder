```typescript
if (isFree) {
  const now = Date.now();

      [pathKey]: used + 1,
      lastUpdatedAt: now,

    await ctx.db.insert("userFreeTier", {
      userId,
      freeExploreUsed: args.path === "exploration" ? 1 : 0,

      createdAt: now,
      lastUpdatedAt: now,
    });
  }
```
